{{- /* 图片渲染钩子：① width/height（治 CLS）② srcset/sizes ③ 图床边缘转换 ④ 首图 fetchpriority=high。尺寸读 data/image-dims.json，构建期零网络依赖。 ⚠️ 新图要重跑 scripts/sync-image-dims.py，否则没有尺寸。 ⚠️ 图床前缀只给 img.hulatu.com；外站图原样输出。 ⚠️ quality=82 是目视下限，别再往 70 以下调（截图会出色带）。 ⚠️ 首图 high 不能同时写 lazy；用 .Page.Store 计数，别混 .Scratch。 ⚠️ 数字是 float64，printf %d 前必须 int()，否则产物出现 %!d(float64=…)。 ⚠️ 带连字符的 data key 用 index hugo.Data "image-dims"；with 会重绑 .， 所以先把 $dest 存下来。 ⚠️ overlay 是空壳，放大用 figure 的 --zoom-src 背景图（custom.css 第 26 段）。 */ -}} {{- $dest := .Destination -}} {{- $alt := .Text -}} {{- $id := printf "img-%s" (substr (md5 $dest) 0 8) -}} {{- $zoom := $.Page.Site.Params.imageZoom -}} {{- /* ① 原图尺寸 */ -}} {{- $w := 0 -}} {{- $dim := "" -}} {{- with index hugo.Data "image-dims" -}} {{- with index . $dest -}} {{- $w = int (index . 0) -}} {{- $dim = printf ` width="%d" height="%d"` (int (index . 0)) (int (index . 1)) -}} {{- end -}} {{- end -}} {{- /* ② 边缘转换 + srcset。只有图床的图才拼前缀，外站图原样输出。 */ -}} {{- $src := $dest -}} {{- /* ⚠️ 必须 safeHTMLAttr：$srcsetAttr 插在属性名位置，普通空串会被转成 zgotmplz。 */ -}} {{- $srcsetAttr := "" | safeHTMLAttr -}} {{- if strings.HasPrefix $dest "https://img.hulatu.com/" -}} {{- $path := strings.TrimPrefix "https://img.hulatu.com/" $dest -}} {{- $fmtW := "https://img.hulatu.com/cdn-cgi/image/width=%d,format=auto,quality=82/%s" -}} {{- $fmtF := "https://img.hulatu.com/cdn-cgi/image/format=auto,quality=82/%s" -}} {{- if and $w (le $w 720) -}} {{- /* 原图 ≤720px：只转格式，不缩宽（二次缩小会糊）。 */ -}} {{- $src = printf $fmtF $path -}} {{- else -}} {{- /* 候选宽度只保留 ≤ 原图宽的档位，避免向图床要一张会被放大回去的图。 $w 为 0（尺寸表里没这条）时全保留，退化成三档。 */ -}} {{- $cands := slice -}} {{- range (slice 400 720 1080) -}} {{- if or (eq $w 0) (le . $w) -}}{{- $cands = $cands | append . -}}{{- end -}} {{- end -}} {{- if ge (len $cands) 2 -}} {{- $parts := slice -}} {{- range $cands -}} {{- $parts = $parts | append (printf "%s %dw" (printf $fmtW . $path) .) -}} {{- end -}} {{- $srcsetAttr = printf ` srcset="%s" sizes="(max-width: 760px) calc(100vw - 40px), 680px"` (delimit $parts ", ") | safeHTMLAttr -}} {{- /* src 取最大可用档；内容区 680px，DPR2 由浏览器选 1080。 */ -}} {{- $src = printf $fmtW (index $cands (sub (len $cands) 1)) $path -}} {{- else if eq (len $cands) 1 -}} {{- /* 防御分支：候选表只剩一档时兜底。 */ -}} {{- $src = printf $fmtW (index $cands 0) $path -}} {{- end -}} {{- end -}} {{- end -}} {{- /* ③ 首图 high、其余 lazy。两者互斥，别同时写。 */ -}} {{- $loadAttr := ` loading="lazy" decoding="async"` -}} {{- if not (.Page.Store.Get "imgFirst") -}} {{- $loadAttr = ` fetchpriority="high" decoding="async"` -}} {{- .Page.Store.Set "imgFirst" true -}} {{- end -}} {{- /* --zoom-src 用原图 URL；含单引号就放弃（遮罩仍在，只是不放大）。 */ -}} {{- $zoomStyle := "" -}} {{- if and $zoom (not (strings.Contains $dest "'")) -}} {{- $zoomStyle = printf ` style="--zoom-src:url('%s')"` $dest | safeHTMLAttr -}} {{- end -}}
