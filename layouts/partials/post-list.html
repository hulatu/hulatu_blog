    本身；标题、计数行由调用方负责。 按年分组由 site.Params.groupByYear 控制，grouped/ungrouped 类名与上游一致。 */ -}} {{- $pages := .pages -}} {{- $empty := .empty | default "这里还没有内容。" -}} {{- $group := site.Params.groupByYear -}} {{- $desc := .desc -}} {{- /* 先按年计数，给年份标题后面的「N 篇」用。直接用 .Date.Year 做 where 的键 不可靠（Year 是方法不是字段），所以手动累计。 */ -}} {{- $yearCounts := dict -}} {{- if $group -}} {{- range $pages -}} {{- $y := printf "%d" .Date.Year -}} {{- $yearCounts = merge $yearCounts (dict $y (add (index $yearCounts $y | default 0) 1)) -}} {{- end -}} {{- end -}}

      • {{- else -}}
